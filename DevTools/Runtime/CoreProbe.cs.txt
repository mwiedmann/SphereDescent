using System.Collections;
using System.Reflection;
using UnityEngine;
using UnityEngine.InputSystem;
using UnityEngine.InputSystem.LowLevel;

namespace SphereDescent
{
    public class PlayProbe : MonoBehaviour
    {
        public static bool Done;

        void Start() => StartCoroutine(Run());

        static T Get<T>(object o, string f) => (T)o.GetType().GetField(f, BindingFlags.NonPublic | BindingFlags.Instance).GetValue(o);

        IEnumerator Press(Key key = Key.Space)
        {
            InputSystem.settings.backgroundBehavior = InputSettings.BackgroundBehavior.IgnoreFocus;
            InputSystem.settings.editorInputBehaviorInPlayMode = InputSettings.EditorInputBehaviorInPlayMode.AllDeviceInputAlwaysGoesToGameView;
            if (!Keyboard.current.enabled) InputSystem.EnableDevice(Keyboard.current);
            InputSystem.QueueStateEvent(Keyboard.current, new KeyboardState(key));
            Debug.Log($"[Probe] after manual update: {Keyboard.current.spaceKey.isPressed} enabled {Keyboard.current.enabled} mode {InputSystem.settings.updateMode}");
            yield return null;
            Debug.Log($"[Probe] space pressed {Keyboard.current.spaceKey.isPressed} any {GameInput.ContinuePressed()}");
            yield return null;
            InputSystem.QueueStateEvent(Keyboard.current, new KeyboardState());
            yield return null;
        }

        IEnumerator Run()
        {
            Application.runInBackground = true;
            if (Keyboard.current == null) InputSystem.AddDevice<Keyboard>();
            yield return new WaitForSeconds(1f);
            var gm = GameManager.Instance;
            while (gm.Arriving) yield return null; // the world's opening meteor arrival
            var ball = FindFirstObjectByType<BallController>();
            var rb = ball.GetComponent<Rigidbody>();
            var level = FindFirstObjectByType<LevelInstance>();
            var core = level.Layout.Core;
            float G = core.GroundRadius;
            ball.Teleport(level.Layout.Holes[level.Layout.Sections.Count - 1].Direction * 36f);
            yield return new WaitForSeconds(2.5f);
            var w = gm.Scores.Current;
            Debug.Log($"[Probe] landed: layer {ball.CurrentLayer} grounded {ball.IsGrounded} r {ball.transform.position.magnitude:F2} G {G:F2} section {Get<int>(gm, "section")} coreIndex {w.CoreIndex} possible {w.Possible[w.CoreIndex]}");
            Vector3 landedAt = ball.transform.position;
            // No lava rain before the walls have been reached.
            float quiet = Time.time;
            bool early = false;
            while (Time.time - quiet < 6f) { if (GameObject.Find("Blob Shadow") != null) early = true; yield return null; }
            Debug.Log($"[Probe] rain before reaching the walls: {early}");
            // Reaching the plaza: the energy ball's introduction plays, and a button only skips it after a second.
            var arenaI = FindFirstObjectByType<CoreArena>();
            Vector3 plazaDir = core.Paths[0][0];
            rb.position = plazaDir * (G + 0.6f); rb.linearVelocity = Vector3.zero;
            yield return new WaitForSeconds(0.4f);
            Debug.Log($"[Probe] intro playing {Get<float>(gm, "introStartedAt") >= 0f}, caption '{Get<string>(gm, "caption")}', frozen {Get<bool>(ball, "frozen")}, orbit {Camera.main.GetComponent<OrbitCamera>().enabled}");
            yield return new WaitForSeconds(2.2f);
            Debug.Log($"[Probe] intro after 2.6s: caption '{Get<string>(gm, "caption")}', camera {Vector3.Distance(Camera.main.transform.position, arenaI.orb.body.position):F1} from the energy ball");
            // Watch the whole introduction: the camera must stay above the ground the whole way.
            float lowest = float.MaxValue, introT = Time.time;
            string steps = "";
            string lastCaption = null;
            while (Get<float>(gm, "introStartedAt") >= 0f && Time.time - introT < 30f)
            {
                Vector3 c = Camera.main.transform.position;
                lowest = Mathf.Min(lowest, c.magnitude - core.SurfaceRadius(c.normalized));
                string cap = Get<string>(gm, "caption");
                if (cap != lastCaption) { steps += $" [{Time.time - introT + 2.6f:F1}s] {cap}"; lastCaption = cap; }
                yield return null;
            }
            Debug.Log($"[Probe] full intro {Time.time - introT + 2.6f:F1}s, camera lowest {lowest:F1} above the ground;{steps}");
            Vector3 before = ball.transform.position;
            yield return Press(Key.W);
            yield return Press(Key.UpArrow);
            yield return new WaitForSeconds(0.5f);
            Debug.Log($"[Probe] steering keys during intro: still playing {Get<float>(gm, "introStartedAt") >= 0f}, ball moved {Vector3.Distance(before, ball.transform.position):F2}");
            yield return Press();
            Debug.Log($"[Probe] intro skipped: {Get<float>(gm, "introStartedAt") < 0f}, done {arenaI.IntroDone}, frozen {Get<bool>(ball, "frozen")}, orbit {Camera.main.GetComponent<OrbitCamera>().enabled}");
            // Sitting still on the core: a lava blob should land on the ball within a few seconds.
            float sat = Time.time;
            bool sawShadow = false;
            while (!ball.IsDestroyed && Time.time - sat < 7f)
            {
                if (GameObject.Find("Blob Shadow") != null) sawShadow = true;
                yield return null;
            }
            Debug.Log($"[Probe] lava rain: hit {ball.IsDestroyed} after {Time.time - sat:F1}s, shadow seen {sawShadow}, orb height {FindFirstObjectByType<EnergyOrb>().body.localPosition.y:F1}");
            while (ball.IsDestroyed) yield return null;
            // Rolling steadily: the blob should be aimed ahead, where the ball is going.
            float rolled = Time.time;
            int deaths = 0;
            Vector3 lastAim = Vector3.zero;
            while (Time.time - rolled < 8f)
            {
                Vector3 up = ball.transform.position.normalized;
                Vector3 dir = Vector3.ProjectOnPlane(core.Cage, up).normalized;
                rb.linearVelocity = dir * 4f + up * Vector3.Dot(rb.linearVelocity, up);
                var shadow = GameObject.Find("Blob Shadow");
                if (shadow != null && lastAim == Vector3.zero)
                {
                    var bounds = shadow.GetComponent<MeshRenderer>().bounds.center;
                    lastAim = bounds;
                    Debug.Log($"[Probe] blob aimed {Vector3.Distance(bounds, ball.transform.position):F1} from the ball, ahead {Vector3.Dot(bounds - ball.transform.position, dir):F1}");
                }
                if (ball.IsDestroyed) { deaths++; while (ball.IsDestroyed) yield return null; }
                yield return null;
            }
            Debug.Log($"[Probe] rolling at 4: hit {deaths} times in 8s");
            // Aimed into the walls, and out over the lava: each blob must still come down on open path.
            var arenaA = FindFirstObjectByType<CoreArena>();
            for (int trial = 0; trial < 6; trial++)
            {
                while (ball.IsDestroyed) yield return null;
                bool towardWalls = trial % 2 == 0;
                Vector3 start, heading;
                if (towardWalls)
                {
                    Vector3 side = Vector3.ProjectOnPlane(core.Landing, core.Cage).normalized;
                    start = (core.Cage + side * ((core.RingRadii[core.RingRadii.Length - 1] + 2f) / G)).normalized;
                    heading = Vector3.ProjectOnPlane(core.Cage - start, start).normalized;
                }
                else
                {
                    var path = core.Paths[1 + trial % (core.Paths.Count - 1)];
                    start = path[path.Count / 2];
                    Vector3 along = path[path.Count / 2 + 1] - start;
                    heading = Vector3.Cross(start, along).normalized; // straight off the side of the path
                }
                rb.position = start * (core.SurfaceRadius(start) + 0.55f);
                yield return new WaitForFixedUpdate();
                foreach (var old2 in GameObject.FindObjectsByType<LavaBlob>(FindObjectsSortMode.None)) Destroy(old2.gameObject);
                yield return null;
                GameObject shadow = null;
                float t1 = Time.time;
                while (shadow == null && Time.time - t1 < 6f)
                {
                    Vector3 up = ball.transform.position.normalized;
                    rb.position = start * (core.SurfaceRadius(start) + 0.55f);
                    rb.linearVelocity = Vector3.ProjectOnPlane(heading, up).normalized * 7f;
                    yield return new WaitForFixedUpdate();
                    shadow = GameObject.Find("Blob Shadow");
                }
                if (shadow == null) { Debug.Log("[Probe] aim trial: no blob"); continue; }
                Vector3 c = shadow.GetComponent<MeshFilter>().sharedMesh.vertices[0].normalized;
                Debug.Log($"[Probe] aim trial {(towardWalls ? "walls" : "lava")}: lands {core.OutsideGround(c):F1} outside the ground (negative = on it), {Vector3.Angle(c, core.Cage) * Mathf.Deg2Rad * G:F1} from the middle, outer wall {core.RingRadii[core.RingRadii.Length - 1]}");
                rb.linearVelocity = Vector3.zero;
            }
            ball.Teleport(landedAt);
            yield return new WaitForFixedUpdate();

            // Find lava near the landing and drop in.
            Vector3 lava = Vector3.zero;
            for (float d = 6f; d < 40f && lava == Vector3.zero; d += 0.5f)
                for (int a = 0; a < 36; a++)
                {
                    Vector3 side = Quaternion.AngleAxis(a * 10f, core.Landing) * Vector3.ProjectOnPlane(Vector3.up, core.Landing).normalized;
                    Vector3 dir = (core.Landing + side * d / G).normalized;
                    if (core.SurfaceRadius(dir) < core.LavaRadius - 1f) { lava = dir; break; }
                }
            rb.position = lava * (core.LavaRadius + 2f); rb.linearVelocity = Vector3.zero;
            float t0 = Time.time;
            while (!ball.IsDestroyed && Time.time - t0 < 3f) yield return null;
            Debug.Log($"[Probe] lava: destroyed {ball.IsDestroyed} after {Time.time - t0:F2}s, r now {ball.transform.position.magnitude:F2}, lava {core.LavaRadius}, surface there {core.SurfaceRadius(lava):F2}, layer {ball.CurrentLayer}, kinematic {rb.isKinematic}");
            yield return new WaitForSeconds(1.6f);
            Debug.Log($"[Probe] respawned {Vector3.Distance(ball.transform.position, landedAt):F2} from landing, layer {ball.CurrentLayer}");

            var arena = FindFirstObjectByType<CoreArena>();
            var order = new System.Collections.Generic.List<int>();
            for (int i = 0; i < arena.barriers.Length; i++) { int k = i; arena.barriers[i].Broke += b => order.Add(k); }
            foreach (var p in core.Pads)
            {
                rb.position = p * (core.SurfaceRadius(p) + 0.6f); rb.linearVelocity = Vector3.zero;
                yield return new WaitForSeconds(0.4f);
            }
            Debug.Log($"[Probe] pads pressed, core earned {w.Earned[w.CoreIndex]}, walls left {arena.Remaining}");
            rb.position = core.Cage * (G + 0.6f) + Vector3.Cross(core.Cage, Vector3.up).normalized * 9.5f;
            t0 = Time.time;
            while (arena.Remaining > 0 && Time.time - t0 < 30f) yield return null;
            Debug.Log($"[Probe] break order (0 = innermost): {string.Join(",", order)}"); Debug.Log($"[Probe] walls all down after {Time.time - t0:F1}s: {arena.Remaining} left");
            foreach (var c in core.Cubes)
            {
                rb.position = c * (core.SurfaceRadius(c) + 0.9f); rb.linearVelocity = Vector3.zero;
                yield return new WaitForSeconds(0.25f);
            }
            Debug.Log($"[Probe] cubes, core earned {w.Earned[w.CoreIndex]}");
            float fps0 = Time.frameCount; float ft = Time.realtimeSinceStartup;
            yield return new WaitForSeconds(2f);
            Debug.Log($"[Probe] fps on core ~{(Time.frameCount - fps0) / (Time.realtimeSinceStartup - ft):F0}");

            while (ball.IsDestroyed) yield return null;
            yield return new WaitForSeconds(0.3f);
            rb.position = core.Cage * (G + 0.6f); rb.linearVelocity = Vector3.zero;
            yield return new WaitForSeconds(0.5f);
            Debug.Log($"[Probe] at orb: destroyed {ball.IsDestroyed} dist {Vector3.Distance(ball.transform.position, arena.orb.body.position):F2} allBroken {arena.AllBroken} r {ball.transform.position.magnitude:F2} broken {string.Join(",", System.Array.ConvertAll(arena.barriers, b => b.Broken))} walls active {string.Join(",", System.Array.ConvertAll(arena.barriers, b => b.wall.activeSelf))}");
            Debug.Log($"[Probe] orb freed {arena.orb.Freed}, thanks {Get<float>(gm, "thanksShownAt"):F1}, core earned {w.Earned[w.CoreIndex]}");
            yield return Press();
            yield return new WaitForSeconds(0.5f);
            Debug.Log($"[Probe] early press ignored: thanks still {Get<float>(gm, "thanksShownAt") >= 0f}");
            yield return new WaitForSeconds(3f);
            yield return Press();
            yield return new WaitForSeconds(0.5f);
            Debug.Log($"[Probe] thanks gone {Get<float>(gm, "thanksShownAt") < 0f}; orbit enabled {Camera.main.GetComponent<OrbitCamera>().enabled}");
            yield return new WaitForSeconds(4.5f);
            Debug.Log($"[Probe] summary page {Get<int>(gm, "summaryPage")}, core finished {w.Finished[w.CoreIndex]}, world banked {w.Banked}/{w.PossibleSoFar}, label {w.Label(w.CoreIndex)}");
            yield return new WaitForSeconds(3.2f);
            yield return Press();
            yield return new WaitForSeconds(2f);
            Debug.Log($"[Probe] now world {gm.World}, worlds scored {gm.Scores.Worlds.Count}, ball layer {ball.CurrentLayer}");
            Done = true;
        }
    }
}
